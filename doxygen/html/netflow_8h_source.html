250static inline void swap_endianness(void *value, size_t len) {
251 switch (len) {
252 case 2: {
253 uint16_t tmp16 = *(uint16_t *) value;
254 tmp16 = swap_endian_16(tmp16);
255 memcpy(value, &tmp16, sizeof(uint16_t));
256 break;
257 }
258 case 4: {
259 uint32_t tmp32 = *(uint32_t *) value;
260 tmp32 = swap_endian_32(tmp32);
261 memcpy(value, &tmp32, sizeof(uint32_t));
262 break;
263 }
264 case 8: {
265 uint64_t tmp64 = *(uint64_t *) value;
266 tmp64 = swap_endian_64(tmp64);
267 memcpy(value, &tmp64, sizeof(uint64_t));
268 break;
269 }
270 case 16: {
271 uint128_t tmp128 = *(uint128_t *) value;
272 tmp128 = swap_endian_128(tmp128);
273 memcpy(value, &tmp128, sizeof(uint128_t));
274 break;
275 }
276 }
277}
